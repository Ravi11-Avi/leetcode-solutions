class Solution {
    public int reverseBits(int n) {
        String bits  =  Integer.toBinaryString(n);
        String paddedBits = String.format("%32s", bits).replace(' ', '0');
        String news =  new StringBuilder(paddedBits).reverse().toString();
        int ab =  Integer.parseInt(news, 2);
        return ab;
    }
    
}
