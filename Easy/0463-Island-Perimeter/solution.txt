class Solution {
    public int islandPerimeter(int[][] grid) {

        boolean founffirst =  false ;
        int ans = 0;
        for( int i = 0 ; i< grid.length ; i++){
            for (int j = 0 ; j < grid[0].length ; j++){

                if(grid[i][j]== 1){
                      if (!founffirst){
                        ans +=4;
                        founffirst = true;
                    }else{
                        ans +=2;
                    }
                }
                  
            }
        }

        return ans;
    }
}
