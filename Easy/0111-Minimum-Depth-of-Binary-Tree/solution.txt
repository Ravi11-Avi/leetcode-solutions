/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    public class Node{
        TreeNode node;
        int depth;

        public Node(TreeNode node, int depth){
            this.node = node;
            this.depth = depth;
        }
    }
    public int minDepth(TreeNode root) {
        if (root ==  null) return 0;

        Queue< Node> q =  new LinkedList<>();
        q.offer(new Node(root,1));

        while (!q.isEmpty()){
            Node curr  =  q.poll();
            TreeNode currT =  curr.node;

        

        if (currT.left == null && currT.right ==  null ) return curr.depth;

        if (currT.left!=null)q.offer(new Node(currT.left, curr.depth+1));
        if (currT.right!=null)q.offer(new Node(currT.right, curr.depth+1));
        }

        return 0;
    }
}
