class Solution {
    public String reverseStr(String s, int k) {
    
    char[] ab = s.toCharArray();


    for (int i = 0 ; i < ab.length ; i += 2*k){

        int l  =  i ;
        int r = Math.min(i+k -1, ab.length-1) ; 

        while(l< r){
            char temp  =  ab[l];
            ab[l]= ab[r];
            ab[r]= temp ;

            l++;
            r--;
        }


        
    }
    

    return new String(ab);
    }
}
