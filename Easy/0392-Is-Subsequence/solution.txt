class Solution {
    public boolean isSubsequence(String s, String t) {
        
        if (s.length()==0) return true;

        int spoint = 0 ;
        int tpointer= 0;


        while (tpointer< t.length()){
            if (s.charAt(spoint)== t.charAt(tpointer))spoint++;
            if (spoint == s.length()-1) return true;

            tpointer++;
        }
        return false;
       
    }
}
