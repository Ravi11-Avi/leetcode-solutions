class Solution {
    public boolean checkDivisibility(int n) {
        int sum =  0;
        int product =  1 ; 
        int nums = n ;

        while (nums> 0){
            int digit =  nums%10;
            sum +=digit;
            product*= digit;

            nums /=10;
        }

        return n%(sum+product)==0;

        
    }
}
