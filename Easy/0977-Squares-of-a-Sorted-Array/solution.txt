class Solution {
    public int[] sortedSquares(int[] nums) {


       PriorityQueue<Integer> pq = new  PriorityQueue<>();

       for (int i : nums){
        pq.add(i*i);
       }

       int[] result = new int[pq.size()];
        int index = 0;

        while (!pq.isEmpty()) {
            result[index++] = pq.poll();
        }

        return result ; 
    }
}
