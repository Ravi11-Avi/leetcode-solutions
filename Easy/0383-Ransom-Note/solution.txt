class Solution {
    public boolean canConstruct(String rn, String mg) {
        int[] ach  =  new int [26];
        int[] abc =  new int[26];

        for (char c : rn.toCharArray())ach[c- 'a']++;
        for (char c : mg.toCharArray())abc[c- 'a']++;

        for(int i = 0 ; i < 26 ; i++){
            if (abc[i]< ach[i])return false;
        }
        return true;
    }
}
