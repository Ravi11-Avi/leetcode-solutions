class Solution {
    public int[] twoSum(int[] nums, int target) {

       HashMap <Integer, Integer> map  = new HashMap<>();

       for (int i =  0 ; i < nums.length ; i++){
        int col =  target - nums[i];

        if (map.containsKey(col)){
            return new int []{map.get(col),i};
        }

        map.put(nums[i],i);
       }

       return new int []{};
        
    }
}
