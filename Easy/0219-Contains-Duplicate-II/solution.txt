class Solution {
    public boolean containsNearbyDuplicate(int[] nums, int k) {
        HashSet<Integer> w = new HashSet<>();
        for(int i = 0 ; i<nums.length  ; i++){
           if(i>k){
            w.remove(nums[i-k-1]);
           }
           if(!w.add(nums[i])){
            w.add(nums[i]);
            return true;
           }
        }
        return false;
    }
}
