class Solution {
    public int[] separateDigits(int[] nums) {
        ArrayList<Integer> aa =  new ArrayList<>();

        for (int i :  nums){
            int curridx= aa.size();
            while(i>0){
                aa.add(curridx,i%10);
                i /=10;
            }
        }


        int[] res =  new int[aa.size()];


        for (int i = 0 ; i < aa.size(); i++){
            res[i]= aa.get(i);
        }

        return res;
    }
}
