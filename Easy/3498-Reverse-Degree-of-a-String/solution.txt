class Solution {
    public int reverseDegree(String s) {
        int sum = 0;
        
        i =1;
        for (int i = 0;i<s.length() ;i++){
            char c =  s.charAt(i);

            sum += (i++)*('z'-c+1);
        }
        return sum;
    }
}
