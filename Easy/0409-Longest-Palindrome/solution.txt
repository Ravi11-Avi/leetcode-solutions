class Solution {
    public int longestPalindrome(String s) {
        int[] ch =  new int[128];


        for(char c : s.toCharArray())ch[c]++;
        int l = 0;
        boolean isadd = false;

        for (int i = 0 ; i < 128 ; i++){
            if (ch[i]%2 ==0){
                l+=ch[i];
            }else{
                l+= ch[i]-1;
                isadd = true;
            }
        }

        if (isadd ) return l+1;
        return l;
        
    }
}
