class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        ArrayList<Integer> arr = new ArrayList<>();
        if (s.length ()< p.length()) return arr;

        int l=0,r=0;
        int[] freq= new int[26];
        for (int i =0; i<p.length() ; i++){
           freq[ p.charAt(i)-'a'] ++;
        }
        int[] window =  new int[26];


        while ( r< s.length() ){
            window[s.charAt(r)-'a']++;

            if (r-l+1 == p.length()){

                if(Arrays.equals(window, freq)){
                    arr.add(l);
                }
                window[s.charAt(l)-'a']--;

                l++;
                
            }
            r++;
            

        }
        return arr;
    }
}
