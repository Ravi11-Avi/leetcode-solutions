class Solution {
    public int maxFrequency(int[] nums, int k) {

        Arrays.sort(nums);

        int maxfre = 0 , l = 0 ;
        long windsum = 0;

        for (int r = 0 ; r < nums.length ; r ++){

            windsum +=  nums[r];

            while((long)nums[r]* (r-l+1) - windsum > k){
                windsum -= nums[l];
                l++;
            }

            maxfre = Math.max(maxfre ,  r-l+1);
        }

        return maxfre;
        


    }
}
