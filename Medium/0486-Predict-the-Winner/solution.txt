class Solution {
    public boolean predictTheWinner(int[] nums) {

        int p1 = 0;
        int p2 = 0;

        for (int i = 0; i <nums.length ; i ++){
            if(i%2 == 0 ){
                p1+=nums[i];
            }else{
                p2 +=nums[i];
            }
        }
        if (p1> p2)return true;

        return false;
        
    }
}
