class Solution {
    public int maxFrequency(int[] nums, int k, int numOperations) {
        int max = 0 ;

        for (int i : nums){
            List<Integer> list =  new ArrayList<>();
            int l = 0 ,r = nums.length -1 ;
            int lowrange =  i-k;
            int highrange = i+k; 

            for (int n :  nums){
                if (n>= lowrange && n<=  highrange)list.add(n);
            }
            int same= 0;

            for (int num : list){
                if (num == i){
                    same++;
                }
            }

            int op  =  Math.min(list.size() - same, numOperations);

            max =  Math.max(max ,  op+ same);

        }

        return max;
    }
}
