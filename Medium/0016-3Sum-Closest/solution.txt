class Solution {
    public int threeSumClosest(int[] nums, int t) {
        
        int closer  = nums[0] + nums[1] + nums[2];
        Arrays.sort(nums);

        for(int i = 0 ; i < nums.length-2; i ++){
            int l = i+1;
            int r = nums.length-1;

            while (l< r){

                int sum  = nums[l]+ nums[i]+ nums[r];
                if (Math.abs(t-closer)>Math.abs(t-sum))closer = Math.abs(t-sum);


                if (sum> t)r--;
                else if (sum< t)l++;
                else return sum;

            }
        }
        return closer;
    }
}
