class Solution {
    public double myPow(double x, int n) {
        long  bi =  n;

        if(n <  0){
            x= 1/x;
            bi = -bi;
        }
        double ans  =  1 ;

        while(bi>0){
            if(bi%2 == 1 ){
                ans *= x;
            }
            x*=x;
            bi /=2;
        }

        return ans;
    }
}
