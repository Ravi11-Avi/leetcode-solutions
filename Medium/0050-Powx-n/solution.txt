//  Problem  : 0050. Pow(x, n)

//  Difficulty : Medium
//  Language : chooseatype
//  Solved   : October 27, 2026

class Solution {
    public double myPow(double x, int n) {
        long bi = n;
        if (bi < 0) {
            x = 1.0 / x;
            bi = -bi;
        }

        double ans = 1.0;
        while (bi > 0) {
            if (bi % 2 == 1) {
                ans *= x;
            }
            x *= x;
            bi /= 2;
        }
        return ans;
    }
}
