class Solution {
    public int[][] findFarmland(int[][] land) {
        int row = land.length;
        int col  = land[0].length;

        List<int[]> result = new ArrayList<>();


        for (int i =  0 ; i <row ; i ++ ){
            for (int j = 0 ; j< col ; j++){
                if (land[i][j]== 1){
                    int x= i, y = j;

                   

                    while (x+1< row && land[x+1][j]==1){
                        x++;
                    }
                    while (y+1< col && land[i][y+1]==1){
                        y++;
                    }

                    result.add(new int[]{i,j,x,y});
                     for(int r= i ;  r<=x; r++){
                        for (int s= j ; s <=y ; s++){
                            land[r][s]=0;
                        }
                    }

                }
            }
        }

        return result.toArray(new int[result.size()][]);

    }
}
