class Solution {
    public int findKthLargest(int[] nums, int k) {

        
        PriorityQueue< Integer> hash = new PriorityQueue();

        for(int i: nums){
            hash.add(i);

            if(hash.size()>k){
                 hash.poll();
            }
        }

        return hash.poll();
    }
}
