class Solution {
    public String countAndSay(int n) {

        if (n ==1) return "1";
        
        String c = "1" ;
        for (int i = 1 ; i < n ; i++){
            StringBuilder sb  =  new StringBuilder();
            int count = 1 ; 

            for (int j = 1 ; j< c.length (); j++){
                if (c.charAt(j)== c.charAt(j-1))count++;
                else{
                    sb.append(count);
                    sb.append(c.charAt(j-1));
                    count =1;
                }
               
            }
            sb.append(count);
            sb.append(c.charAt(c.length()-1));

            c = sb.toString();

            

        }
        return c;
    }
}
