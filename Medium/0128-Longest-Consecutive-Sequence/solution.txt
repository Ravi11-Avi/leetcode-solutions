class Solution {
    public int longestConsecutive(int[] nums) {
        Arrays.sort(nums);
        int curcount = 1;
        int max =  1;

        for(int i = 0 ; i < nums.length-1 ; i++){
            if(nums[i]==nums[i+1])continue;
            if (nums[i]==nums[i+1]-1){
                curcount++;
            }else{
                curcount=1;
            }
            if (max< curcount)max=curcount;
        }
        return max;
    }
}
