class Solution {
    public long countCommas(long n) {
    
       long ans = 0; 

       for (long i = 1000 ; i <= n ;){
            ans+= (n-i+1);
            
            if(i> Long.MAX_VALUE/1000) break;

            i*=1000;
       }

       

       
    return ans;
    }

}
