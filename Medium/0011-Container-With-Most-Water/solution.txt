class Solution {
    public int maxArea(int[] height) {
        int i = 0;
        int j =  height.length -1; 
        int max= 0 ;

        while (i< j){
            int wid = j-i;

            int cm =  Math.min(height[i], height[j])* wid;

            if (max< cm) max= cm;

            if (height[i]< height[j]) i++;
            else j--;
       }

       return max;
    }
}
