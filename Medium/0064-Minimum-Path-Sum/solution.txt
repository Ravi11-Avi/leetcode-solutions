class Solution {
    public int minPathSum(int[][] grid) {
        int row =  grid.length;
        int col =  grid[0].length;
        int sum = 0 ;

        for (int i = 0 ; i < col ; i ++){
            sum+= grid[0][i];
        }
        for(int i =1 ; i < row ; i++){
            sum+= grid[i][col-1];
        }
        return sum;
    }
}
