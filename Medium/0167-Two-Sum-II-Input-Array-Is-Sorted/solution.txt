class Solution {
    public int[] twoSum(int[] num, int target) {
        HashMap <Integer, Integer> map =  new HashMap<>();
       

        for (int i = 0 ; i< num.length ; i++ ){
            int c =  target-  num[i];

            if (map.containsKey(c)){
                return new int[]{map.get(c)+1,i+1};
            }

            map.put(num[i], i);
        }

        return new int[]{};
    }
}
