class Solution {
    public int longestSubarray(int[] nums) {
        int l = 0 , count = 0  ,  max0freq = 0 ; 

        for(int  r= 0 ;r< nums.length; r++){
            if (nums[r]==0)max0freq++;

            while(max0freq > 1){
                if (nums[l]==0)max0freq-- ; 
                l++;
            }

            count= Math.max(count,  r-l);
        }
        return count ;
    }
}
