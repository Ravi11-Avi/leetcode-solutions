class Solution {
    public String reverseWords(String s) {
        
        String[] word =  s.trim().split(" ");

        int l= 0 ;
        int r= word.length -1;

        while (l< r){
            String temp  = word[l];
            word[l]= word[r];
            word[r] = temp;
            
            l++;
            r--;

        }
        StringBuilder sb =  new StringBuilder();

        for(int  i = 0  ; i < word.length -1 ; i ++){
            sb.append (word[i]);
            sb.append(" ");
        }
        sb.append(word[word.length -1]);


        return sb.toString();
    }
}
