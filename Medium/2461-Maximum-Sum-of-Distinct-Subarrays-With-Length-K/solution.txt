class Solution {
    public long maximumSubarraySum(int[] nums, int k) {

        HashSet<Integer> set =  new HashSet<>();
        int  l = 0  ;
        long currentsum = 0 , maxres  = 0 ; 

        for (int r= 0 ; r < nums .length  ; r ++){

            while (set.contains(nums[r])){
                set.remove(nums[l]);
                currentsum -= nums[l];
                l++;

            }
            set.add(nums[r]);
            currentsum += nums[r]; 

            if (r-l+1 == k){
                maxres =  Math.max(maxres, currentsum);               
                set.remove(nums[l]);
                currentsum -= nums[l];
                l++;


            }

            

        }
        return maxres;
        
    }
}
