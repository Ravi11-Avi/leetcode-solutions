class Solution {
    public int longestSubarray(int[] nums, int limit) {
         ArrayDeque<Integer> inc   = new ArrayDeque<>();
         ArrayDeque<Integer> dec   = new ArrayDeque<>();

         int res = 0 ,  l = 0 ;

         for (int r= 0 ; r < nums.length ; r++){
            while (!inc.isEmpty()&& inc.getLast()> nums[r])inc.removeLast();      
            while (!dec.isEmpty()&& dec.getLast()< nums[r])dec.removeLast();
            
            inc.addLast(nums[r]);
            dec.addLast(nums[r]);

            while(dec.getFirst()- inc.getFirst()> limit){
                if(inc.getFirst()== nums[l])inc.removeFirst();
                if(dec.getFirst()== nums[l])dec.removeFirst();
                
                l++;
            }

            res =  Math.max(res, r-l+1);
         }
         return res;
    }
}
