class Solution {
    public int scoreOfParentheses(String s) {
        Stack<Integer> stack =  new Stack<>();


      int result = 0 ;

      for (char c: s.toCharArray()){
        if(c=='('){
            stack.push(result);
            result = 0 ;
            
        }else{
            int insc =  result;
            int dbl =  Math.max(2* insc , 1);

            result = dbl + stack.pop();
        }
      } 

      return result; 
    }
}
