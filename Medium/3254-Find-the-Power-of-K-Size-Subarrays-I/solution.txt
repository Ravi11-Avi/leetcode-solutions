class Solution {
    public int[] resultsArray(int[] nums, int k) {
        int l = 0 ,r = k ; 
        int[] results = new int[nums.length - k + 1];
        int idx = 0;

        while (r<= nums.length){
            boolean ca=  true;
            for(int i = l+1 ; i < r ; i++){
                if (nums[i-1]+1!= nums[i]){
                    ca= false;
                    break;
                }
            }
            if (ca){
                results[idx++]= nums[r-1];
            }else{
                results[idx++]= -1;
            }
            l++;
            r++;
        }
        return results;
    }
}
