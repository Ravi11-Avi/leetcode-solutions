class Solution {
    public int minInsertions(String s) {
        
        Stack<Integer> stack = new Stack<>();
        int need =0;
        int ins =  0 ;


        for (char c: s.toCharArray()){
            if (c=='('){
                
                if (need%2!=0){
                    ins++;
                    need--;
                }
                need+=2;
                
            }
            else{
                need--;
                if (need<0){
                        need=1;
                        ins++;
                }
            }

        }
             return need+ ins;

    }
}
