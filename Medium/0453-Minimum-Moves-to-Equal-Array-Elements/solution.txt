class Solution {
    public int minMoves(int[] nums) {
        long sum  = 0;
        int min= nums[0];

        for (int i :  nums){
            sum+=i;
            if (i> min)min = i;

        }

        return  (int)(Math.abs(sum -( nums.length* min)));
    }
}
