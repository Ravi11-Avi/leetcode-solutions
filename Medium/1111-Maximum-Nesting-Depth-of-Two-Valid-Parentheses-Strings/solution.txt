class Solution {
    public int[] maxDepthAfterSplit(String seq) {

        boolean current = true ; 
        int [] res  =  new int[seq.length()]; 

        for(int i = 0 ; i < seq.length()-1; i++){
            if (seq.charAt(i) =='('){
                current = !current;
                if (current) res[i]= 1;
                else res[i]=0;

            }else  {
                if (current) res[i]= 1;
                else res[i]=0;
                current = !current;
            }
        }

        return res;

    }
}
