class Solution {
    public int shortestSubarray(int[] nums, int k) {
        
        int l = 0 , currentsum= 0 , max= -1;

        for(int r = 0 ; r < nums.length ; r++){
            currentsum +=nums[r];


            while(currentsum> k && l< r){

                if (nums[r]== k) return 1;
                currentsum-= nums[l];
                l++;
            }
                if (r-l+1>=k){
                   max= Math.max(max, r-l+1);

                }


        }
        return max;

    }
}
