class Solution {
    public int sumCounts(int[] nums) {
        int n  =  nums.length;
        long sum= 0;
        long MOD = 1_000_000_007;

        for(int i = 0 ; i< n ;i++){
            HashSet<Integer> set =  new HashSet<>();

            for(int j =i; j< n ; j++){
                set.add(nums[j]);
                long r =  set.size();
                sum= (sum+( r*r))%MOD;

            }
        }
        return (int)sum;
    }
}
