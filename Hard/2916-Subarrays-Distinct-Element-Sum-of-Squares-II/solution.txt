class Solution {
    public int sumCounts(int[] nums) {
        int n  =  nums.length;
        int sum= 0;

        for(int i = 0 ; i< n ;i++){
            HashSet<Integer> set =  new HashSet<>();

            for(int j =i; j< n ; j++){
                set.add(nums[j]);
                int r =  set.size();
                sum+= r*r;

            }
        }
        return sum;
    }
}
